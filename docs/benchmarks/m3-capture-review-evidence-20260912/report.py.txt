    rows.append({'cell':cell,'values':data,'graph_speedup':ratios,'extra_memory_mib':memory,'gates':gates})

import math
for row in rows:
    cell = row['cell']
    graphs = [results[f'G{i}-{cell}-measured'] for i in [1, 2]]
    compact = [results[f'C{i}-{cell}-measured'] for i in [1, 2]]
    row['inactive_row_fraction'] = []
    row['setup_break_even_generated_tokens_vs_compact'] = []
    row['per_request_latency_ms'] = {}
    for g, c in zip(graphs, compact):
        counters = g['capture']['final']
        if not any(counters['fallback_counts'].values()):
            live = sum(int(k) * v for k, v in g['counts']['recurrent_occupancy'].items())
            physical = sum(int(k) * b['counters']['replays'] for k, b in counters['buckets'].items())
            assert physical >= live
            row['inactive_row_fraction'].append(1 - live / physical)
        else:
            row['inactive_row_fraction'].append(None)
        saving = 1 / c['metrics']['generated_tokens_per_second'] - 1 / g['metrics']['generated_tokens_per_second']
        overhead = max(0, g['setup_ns'] - c['setup_ns']) / 1e9
        row['setup_break_even_generated_tokens_vs_compact'].append(math.ceil(overhead / saving) if saving > 0 else None)
    for mode in ['C', 'E', 'G']:
        row['per_request_latency_ms'][mode] = [{key: {f: v[f] / 1e6 for f in ('ttft_ns', 'tpot_ns', 'completion_latency_ns', 'admission_queue_ns')} for key, v in results[f'{mode}{i}-{cell}-measured']['metrics']['per_request'].items()} for i in [1, 2]]
