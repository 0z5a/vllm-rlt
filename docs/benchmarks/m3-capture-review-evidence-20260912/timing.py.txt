def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda: f.read(8 * 1024**2), b''):
            h.update(chunk)
    return h.hexdigest()

def write(path, data):
    Path(path).write_text(json.dumps(data, indent=2, sort_keys=True, allow_nan=False) + '\n')

def read(path):
    return json.loads(Path(path).read_text())

def file_manifest(folder):
    return {str(p.relative_to(folder)): digest(p) for p in sorted(folder.rglob('*'))
            if p.is_file() and '__pycache__' not in p.parts}

def probe():
    import torch
    def blocked(*args, **kwargs):
        raise RuntimeError('CPU probe forbids accelerator initialization')
    torch.cuda._lazy_init = blocked
    torch.cuda.is_available = lambda: False
    torch.cuda.device_count = lambda: 0
    from vllm_lt.worker.recurrent_graph import GraphLimits
    from dataclasses import asdict
    original = read(SOURCE / 'benchmarks/fixtures/ouro-m1.json')
    by_id = {w['workload_id']: w for w in original['workloads']}
    workloads = []
    for name, parent, count in [('single', 'W1', 1), ('batch4', 'W2', 4),
                                 ('mixed8', 'W4', 8), ('batch8', 'W2', 8),
                                 ('prefill512', 'W3', 1)]:
        w = copy.deepcopy(by_id[parent])
        w['workload_id'] = name
        w['requests'] = w['requests'][:count]
        if 'replay' in w:
            w['replay'] = {r['request_id']: w['replay'][r['request_id']] for r in w['requests']}
        if name == 'mixed8':
            for i, r in enumerate(w['requests']):
                length = 8 + 2 * i
                r['max_output_tokens'] = length
                w['replay'][r['request_id']] = {k: v[:length] for k, v in w['replay'][r['request_id']].items()}
        workloads.append(w)
    contract = read(SOURCE / 'benchmarks/fixtures/ouro-m1-contract.json')
    contract['limits']['workload_timeout_s'] = 180
    contract['limits']['total_timeout_s'] = 1200
    contract['limits']['profile_decode_outputs'] = 112
    workers = []
    for mode in ['C', 'E', 'G']:
        workers.append({'id': 'F-' + mode, 'mode': mode, 'phases': ['feasibility'], 'cells': [w['workload_id'] for w in workloads]})
    for worker, mode in [('C1','C'),('E1','E'),('G1','G'),('G2','G'),('E2','E'),('C2','C')]:
        workers.append({'id': worker, 'mode': mode, 'phases': ['warmup','measured'], 'cells': [w['workload_id'] for w in workloads]})
    for mode in ['E','G']:
        workers.append({'id': 'P-' + mode, 'mode': mode, 'phases': ['warmup','profile'], 'cells': ['mixed8']})
    stats = {}
    for w in workloads:
        required = sum(4 * ((len(r['prompt_token_ids']) + r['max_output_tokens'] - 2) // 16 + 1) for r in w['requests'])
        assert required <= 1024
        for r in w['requests']:
            assert all(0 <= t < 49152 for t in r['prompt_token_ids'])
        stats[w['workload_id']] = {'pool_bytes': 6 * 1024**3, 'required_blocks': required,
            'requests': len(w['requests']), 'prompt_lengths': [len(r['prompt_token_ids']) for r in w['requests']],
            'output_lengths': [r['max_output_tokens'] for r in w['requests']]}
    # CPU scheduler simulation verifies the mixed diagnostic window before device work.
    from vllm_lt.benchmarks.replay import ReplayEngine
    from vllm_lt.models import OuroConfig, OuroForCausalLM
    from vllm_lt.config import CacheConfig, SchedulerConfig
    from vllm_lt.sampling_params import SamplingParams
    from vllm_lt.request import Stage
    cfg = read(MODEL / 'config.json')
    small = dict(cfg, hidden_size=8, intermediate_size=16, num_hidden_layers=1,
                 num_attention_heads=1, num_key_value_heads=1, head_dim=8, layer_types=['full_attention'])
    tiny = OuroForCausalLM(OuroConfig.from_dict(small))
    mixed = next(w for w in workloads if w['workload_id'] == 'mixed8')
    engine = ReplayEngine(tiny, replay=mixed['replay'], cache_config=CacheConfig(num_blocks=1024), scheduler_config=SchedulerConfig())
    engine.cache_manager.finalize_token = lambda *a: None
    def simulated(batch):
        return [0 if batch.stage == Stage.CODA else 0.0] * len(batch.items)
    engine.model_runner.execute = simulated
    for r in mixed['requests']:
        engine.add_request(r['request_id'], r['prompt_token_ids'], SamplingParams(**dict(contract['engine']['sampling'], max_tokens=r['max_output_tokens'])))
    window, started, emitted = [], False, 0
    while engine.has_unfinished_requests() and emitted < contract['limits']['profile_decode_outputs']:
        outputs = engine.step()
        b = engine.last_schedule
        started = started or b.stage == Stage.PRELUDE
        if started and b.stage == Stage.RECURRENT:
            n = len(b.items)
            window.append({'live_rows': n, 'depths': [x.request.loops_done for x in b.items], 'bucket': engine.model_runner._decode_layout.select(n, max((len(x.request.prompt_token_ids) + x.request.sampling_params.max_tokens - 2) // 16 + 1 for x in b.items))[0]})
        if started:
            emitted += sum(len(o.token_ids) > 1 for o in outputs)
    assert window and set(x['bucket'] for x in window) == {4, 8, 16}
    assert emitted == contract['limits']['profile_decode_outputs'] and not engine.has_unfinished_requests()
    fallback = next(w for w in workloads if w['workload_id'] == 'prefill512')
    fallback_width = max((len(r['prompt_token_ids']) + r['max_output_tokens'] - 2) // 16 + 1 for r in fallback['requests'])
    assert engine.model_runner._decode_layout.select(1, fallback_width) == (None, 'table_width')
    for request_id in list(engine.scheduler.requests):
        engine.abort_request(request_id)
    del engine, tiny
    plan = {'python_executable': sys.executable, 'source_sha': SHA, 'source_manifest': file_manifest(SOURCE),
        'harness_sha256': digest(__file__), 'model_path': str(MODEL), 'model_config': cfg,
        'model_revision': original['model_revision'], 'model_manifest': file_manifest(MODEL),
        'dependencies': {d.metadata['Name']: d.version for d in importlib.metadata.distributions()},
        'contract': contract, 'graph_limits': asdict(GraphLimits()), 'suite': {'workloads': workloads},
        'workload_stats': stats, 'workers': workers, 'gpu_id': 2,
        'controls': {'affinity': list(range(56,64)), 'memory_node': 1, 'dtype': 'float32',
            'scheduler': 'refill', 'model_load': 'once per worker', 'engine': 'fresh per execution',
            'cache_policy': 'preserve allocator within each worker; no shared cache drops',
            'sampling': 'greedy fixed four loops; mixed8 forces archived token/depth trace after actual model/gate/sampling work'},
        'hypothesis': 'Recurrent graph replay reduces launch overhead enough to improve single-request end-to-end throughput versus compact eager; padded eager diagnoses replay savings.',
        'variables': {'G_vs_E': 'only use_graphs changes', 'G_vs_C': 'graph executor including padding, staging, and routing costs versus default compact executor on the SAME source SHA'},
        'acceptance': 'Both G/C single-request throughput ratios >=1.10, single TTFT <=1.05, other workload throughput ratios >=0.95; nonoverlapping primary ranges; per-pair peak allocated/reserved increase <=512 MiB. Report failures/inconclusive without extra runs.',
        'correctness': 'All output histories/depths match across modes; feasibility actual coda argmax sequences match and first/last coda logits allclose(atol=0.001,rtol=0.0001); finite recurrent and coda outputs; zero retained request KV blocks. This is a bounded smoke/numerical screen; lifecycle is separately planned and measured, not quality qualification.',
        'budget': {'feasibility':15, 'measured':30, 'warmup':32, 'profile':2, 'total_executions':79, 'workers':11, 'deadline_s':1200, 'artifact_cap_bytes':3*1024**3},
        'stop': 'Stop on execution/cleanup/identity/numerical failure or time/artifact limits. No retry or additional measurement. Performance guardrails evaluated after frozen samples.',
        'mixed_profile_cpu_simulation': window, 'fallback_cpu_check': {'cell': 'prefill512', 'table_width': fallback_width, 'expected_reason': 'table_width'},
        'preparation': {'downloads': 0, 'source_copy': 'git archive of frozen PR #15 source to task-owned /tmp directory, excluded from timing', 'environment_copy': 'task-owned copy of the prepared environment; no downloads or upgrades; excluded from timing'},
        'created_utc': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())}
    plan['plan_sha256'] = hashlib.sha256(json.dumps(plan, sort_keys=True).encode()).hexdigest()
    write(ROOT / 'plan.json', plan)
    print(json.dumps({'probe':'passed','cuda_initialized':torch.cuda.is_initialized(), 'budget':plan['budget'], 'stats':stats, 'mixed_profile_buckets':sorted(set(x['bucket'] for x in window)), 'plan_sha256':plan['plan_sha256']}))

def worker(worker_id):
    import torch
    from contextlib import contextmanager
    from vllm_lt.benchmarks import runner
    from vllm_lt.benchmarks.capture_runtime import ExecutionAdapter
    from vllm_lt.benchmarks.profile import Capture, finite_checks
    p = read(ROOT / 'plan.json')
    assert file_manifest(SOURCE) == p['source_manifest']
    assert digest(__file__) == p['harness_sha256']
    assert {d.metadata['Name']: d.version for d in importlib.metadata.distributions()} == p['dependencies']
    assert os.environ.get('CUDA_VISIBLE_DEVICES') == str(p['gpu_id'])
    assert sorted(os.sched_getaffinity(0)) == p['controls']['affinity']
    w = next(w for w in p['workers'] if w['id'] == worker_id)
    assert sys.executable == p['python_executable']
    mode = w['mode']
    folder = RAW / 'results' / 'workers'
    folder.mkdir(parents=True, exist_ok=True)
    manifest = {'worker': w, 'status':'running', 'completed':[]}
    class Adapter:
        def __init__(self):
            self.inner = None if mode == 'C' else ExecutionAdapter(implementation_id='B' if mode == 'G' else 'A', graph_limits=p['graph_limits'])
            self.capture_type = Capture if self.inner is None else self.inner.capture_type
        def prepare(self, engine, run, run_dir):
            self.engine, self.run_dir = engine, run_dir
            if self.inner:
                self.inner.prepare(engine, run, run_dir)
        @contextmanager
        def finite_checks(self, engine, enabled):
            if not enabled:
                yield {}
                return
            samples, choices = {}, []
            original = engine.model.coda
            def coda(*args, **kwargs):
                value = original(*args, **kwargs)
                saved = value.detach().cpu().clone()
                samples.setdefault('first', saved)
                samples['last'] = saved
                choices.append(saved.argmax(-1).tolist())
                return value
            engine.model.coda = coda
            executor = engine.model_runner._decode_executor
            recurrent = executor.recurrent if executor else None
            if recurrent:
                def checked(*args, **kwargs):
                    result = recurrent(*args, **kwargs)
                    assert all(bool(x.isfinite().all().item()) for x in result)
                    return result
                executor.recurrent = checked
            try:
                with finite_checks(engine.model, True) as counts:
                    yield counts
            finally:
                engine.model.coda = original
                if recurrent:
                    executor.recurrent = recurrent
                torch.save(samples, self.run_dir / 'coda-logits.pt')
                write(self.run_dir / 'actual-coda-argmax.json', choices)
        def finish(self, engine, capture, *, setup_ns):
            value = self.inner.finish(engine, capture, setup_ns=setup_ns) if self.inner else {'enabled':False}
            self.engine = None
            return value
        def abort(self, error, run_dir):
            if self.inner:
                self.inner.abort(error, run_dir)
            self.engine = None
    model = None
    adapter = Adapter()
    try:
        runner.configure_process(p['contract'])
        manifest['environment'] = runner.environment()
        manifest['numa_policy'] = subprocess.check_output(['numactl','--show'], text=True)
        assert manifest['environment']['host'] == 'dedicated-developjob-8gpu2-a029z-64896bc8cf-8p2lw'
        model = runner.load_model(p, manifest)
        rows = []
        for cell in w['cells']:
            for phase in w['phases']:
                rows.append({'run_id':f'{worker_id}-{cell}-{phase}', 'cell_id':cell, 'workload_id':cell,
                    'phase':phase,'mode':'refill','implementation_id':'B' if mode == 'G' else 'A',
                    'use_graphs':mode == 'G','configuration':mode,'max_steps':4096,'max_events':100000,
                    'instrumentation':'cpu-cuda-profiler' if phase == 'profile' else 'lightweight-host-events',
                    'controls_sha256':p['plan_sha256'], 'workload_sha256':hashlib.sha256(json.dumps(next(x for x in p['suite']['workloads'] if x['workload_id']==cell),sort_keys=True).encode()).hexdigest()})
        def completed(run_id, result):
            manifest['completed'].append(run_id)
            write(folder / f'{worker_id}.json', manifest)
        runner.run_loaded_rows(model, p, rows, output_dir=RAW/'results', deadline=time.perf_counter_ns()+500*10**9, record_completed=completed, execution_adapter=adapter)
        manifest['status']='complete'
    except BaseException:
        manifest['status']='failed'
        manifest['error']=traceback.format_exc()
        raise
    finally:
        model = None
        adapter = None
        gc.collect()
        if torch.cuda.is_initialized():
            runner.release_device(manifest)
        write(folder / f'{worker_id}.json', manifest)

def compare_feasibility():
    import torch
    p = read(ROOT / 'plan.json')
    comparisons=[]
    for cell in p['workload_stats']:
        paths={m:RAW/'results'/'runs'/f'F-{m}-{cell}-feasibility' for m in ['C','E','G']}
        base=read(paths['C']/'result.json')
        history=lambda r:[(x['request_id'],x['token_ids'],x['exit_depths']) for x in r['requests']]
        for mode in ['E','G']:
            r=read(paths[mode]/'result.json')
            assert history(base)==history(r), (cell,mode,'histories')
            assert read(paths['C']/'actual-coda-argmax.json')==read(paths[mode]/'actual-coda-argmax.json'), (cell,mode,'actual coda argmax')
            a=torch.load(paths['C']/'coda-logits.pt',map_location='cpu',weights_only=True)
            b=torch.load(paths[mode]/'coda-logits.pt',map_location='cpu',weights_only=True)
            errors={k:float((a[k]-b[k]).abs().max()) for k in a}
            assert all(torch.allclose(a[k],b[k],atol=.001,rtol=.0001) for k in a), (cell,mode,errors)
            comparisons.append({'cell':cell,'configuration':mode,'passed':True,'max_abs_coda_logit_errors':errors})
    write(ROOT/'feasibility-comparison.json',comparisons)

def controller():
    p=read(ROOT/'plan.json')
    assert file_manifest(MODEL)==p['model_manifest']
    start=time.monotonic()
    write(ROOT/'reservation-environment.json', {'pid':os.getpid(), 'cuda_visible_devices':os.environ.get('CUDA_VISIBLE_DEVICES'), 'scheduler':json.loads(subprocess.check_output(['gpu','status','--json'],text=True)), 'started_utc':time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime())})
    for i,w in enumerate(p['workers']):
        remaining=p['budget']['deadline_s']-(time.monotonic()-start)
        assert remaining>0
        log=ROOT/f"{w['id']}.log"
        print(f"START {w['id']}",flush=True)
        with log.open('w') as f:
            subprocess.run([sys.executable,__file__,'worker','--worker-id',w['id']],stdout=f,stderr=subprocess.STDOUT,check=True,timeout=min(540,remaining))
        print(f"DONE {w['id']}",flush=True)
        if i==2:
            compare_feasibility()
            print('Feasibility comparison passed across all three configurations.',flush=True)
        assert sum(x.stat().st_size for x in RAW.rglob('*') if x.is_file()) < p['budget']['artifact_cap_bytes']
    write(ROOT/'completed.json',{'elapsed_s':time.monotonic()-start,'executions':p['budget']['total_executions'],'status':'complete'})

if __name__=='__main__':
    parser=argparse.ArgumentParser()
    parser.add_argument('action',choices=['probe','worker','run'])
    parser.add_argument('--worker-id')
    args=parser.parse_args()
    if args.action=='probe': probe()
    elif args.action=='worker': worker(args.worker_id)
    else: controller()
