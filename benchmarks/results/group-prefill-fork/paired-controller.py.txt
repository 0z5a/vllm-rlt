import hashlib
import json
import os
import subprocess
from pathlib import Path

root = Path(__file__).resolve().parents[1]
repo = root / "rlt"
parent = root / "baselines/rlt-group-fork-parent"
output = root / "evidence/group-fork-clean-cpu"
output.mkdir(exist_ok=False)
python = "/Users/0z5a/Documents/infra/looped-grpo-integration-20261003/.venv/bin/python"
driver = repo / "benchmarks/benchmark_group_fork.py"
source = {str(p.relative_to(repo)): hashlib.sha256(p.read_bytes()).hexdigest()
          for p in sorted((repo / "vllm_rlt").rglob("*.py"))}
(output / "candidate-source.json").write_text(json.dumps(source, indent=2) + "\n")
receipts = []
for seed in (31, 32, 33):
    for label, fork in (("parent-a1", False), ("fork-b1", True), ("fork-b2", True), ("parent-a2", False)):
        for name, digest in source.items():
            assert hashlib.sha256((repo / name).read_bytes()).hexdigest() == digest, name
        identity = f"s{seed}-{label}"
        env = os.environ | {
            "PYTHONPATH": str(repo if fork else parent), "OMP_NUM_THREADS": "1", "PYTHONDONTWRITEBYTECODE": "1",
            "FROZENLIST_NO_EXTENSIONS": "1", "MULTIDICT_NO_EXTENSIONS": "1", "AIOHTTP_NO_EXTENSIONS": "1",
            "YARL_NO_EXTENSIONS": "1",
        }
        command = [python, "-u", str(root / "bench/resident_pytest.py"), "--script", str(driver),
                   "--seed", str(seed), "--label", label, "--output", str(output / f"{identity}.jsonl")]
        if fork:
            command.append("--fork")
        print(f"Starting {identity}", flush=True)
        with (output / f"{identity}.log").open("x") as stream:
            process = subprocess.run(command, cwd=repo, env=env, stdout=stream, stderr=subprocess.STDOUT)
        receipts.append({"identity": identity, "returncode": process.returncode})
        (output / "processes.json").write_text(json.dumps(receipts, indent=2) + "\n")
        print(f"Finished {identity}: {process.returncode}", flush=True)
        if process.returncode:
            raise SystemExit(process.returncode)
        rows = [json.loads(line) for line in (output / f"{identity}.jsonl").read_text().splitlines()]
        assert len(rows) == 40
        assert all(row["source_root"] == str(repo if fork else parent) for row in rows)
print("All 12 fresh paired processes completed naturally", flush=True)
