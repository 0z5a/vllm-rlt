"""Run the same real preemption/resume sequence against parent or candidate."""
import inspect
import json
import runpy
from pathlib import Path

from vllm_rlt import CacheConfig, SamplingParams, SchedulerConfig
from vllm_rlt.engine.llm_engine import LLMEngine
from vllm_rlt.request import Stage

root = Path(__file__).resolve().parents[1]
make_model = runpy.run_path(str(root / "rlt/benchmarks/benchmark_group_fork.py"))["make_model"]
engine = LLMEngine(
    make_model("ouro", 31),
    cache_config=CacheConfig(256, 8),
    scheduler_config=SchedulerConfig(enable_preemption=True, prefill_chunk_size=3),
)
engine.add_request("0", [2, 3, 4], SamplingParams(max_tokens=4, ignore_eos=True, min_loops=4))
for _ in range(100):
    if engine.scheduler.requests["0"].stage == Stage.PRELUDE:
        break
    engine.step()
assert engine.scheduler.requests["0"].stage == Stage.PRELUDE
engine.add_request("1", [5], SamplingParams(max_tokens=4, ignore_eos=True, min_loops=4))
engine.scheduler.selected_request_ids.clear()
assert engine.preemption.preempt(engine.scheduler.requests["1"])
engine.abort_request("1")
assert engine.scheduler.requests["0"].stage == Stage.WAITING
print(json.dumps({"source_root": str(Path(inspect.getfile(LLMEngine)).resolve().parents[2]),
                  "state_before_resume": "WAITING", "group_fork": "disabled"}), flush=True)
outputs = []
for _ in range(100):
    if not engine.has_unfinished_requests():
        break
    outputs.extend(engine.step(final_only=True))
assert not engine.has_unfinished_requests() and len(outputs) == 1
assert len(outputs[0].token_ids) == 4
assert not engine.cache_manager._allocations and not any(engine.cache_manager._refs)
print("Natural completion: original request resumed from PRELUDE and drained", flush=True)
